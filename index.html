-- =============================================================================
-- SISTEMA MATRI-CLICK: BASE DE DATOS INSTITUCIONAL (ESQUEMAS: auth y colegio)
-- =============================================================================

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE SCHEMA IF NOT EXISTS auth;
CREATE SCHEMA IF NOT EXISTS colegio;

-- TABLA: auth.users
CREATE TABLE auth.users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nombre VARCHAR(50) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    email_verified_at TIMESTAMP WITH TIME ZONE NULL,
    password VARCHAR(255) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- TABLA: colegio.sedes
CREATE TABLE colegio.sedes (
    id_sede SERIAL PRIMARY KEY,
    nombre_sede VARCHAR(100) NOT NULL,
    direccion VARCHAR(150) NOT NULL,
    telefono VARCHAR(20),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- TABLA: colegio.grados
CREATE TABLE colegio.grados (
    id_grado SERIAL PRIMARY KEY,
    nombre_grado VARCHAR(50) NOT NULL,
    nivel_educativo VARCHAR(50) NOT NULL
);

-- TABLA: colegio.periodos_academicos
CREATE TABLE colegio.periodos_academicos (
    id_periodo SERIAL PRIMARY KEY,
    nombre_periodo VARCHAR(50) NOT NULL,
    fecha_inicio DATE NOT NULL,
    fecha_fin DATE NOT NULL,
    estado VARCHAR(20) DEFAULT 'Activo'
);

-- TABLA: colegio.profesores
CREATE TABLE colegio.profesores (
    id_profesor SERIAL PRIMARY KEY,
    user_id UUID REFERENCES auth.users(id) ON DELETE SET NULL,
    documento VARCHAR(20) NOT NULL UNIQUE,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    telefono VARCHAR(20),
    correo_electronico VARCHAR(100) NOT NULL,
    estado_profesor VARCHAR(20) DEFAULT 'Activo'
);

-- TABLA: colegio.acudientes
CREATE TABLE colegio.acudientes (
    id_acudiente SERIAL PRIMARY KEY,
    user_id UUID REFERENCES auth.users(id) ON DELETE SET NULL,
    tipo_documento VARCHAR(20) NOT NULL,
    documento VARCHAR(20) NOT NULL UNIQUE,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    correo_electronico VARCHAR(100),
    direccion VARCHAR(150),
    parentesco VARCHAR(50) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- TABLA: colegio.grupos
CREATE TABLE colegio.grupos (
    id_grupo SERIAL PRIMARY KEY,
    id_grado INT NOT NULL REFERENCES colegio.grados(id_grado),
    id_sede INT NOT NULL REFERENCES colegio.sedes(id_sede),
    nombre_grupo VARCHAR(50) NOT NULL,
    cupo_maximo INT NOT NULL DEFAULT 35,
    id_director_grupo INT REFERENCES colegio.profesores(id_profesor)
);

-- TABLA: colegio.estudiantes
CREATE TABLE colegio.estudiantes (
    id_estudiante SERIAL PRIMARY KEY,
    user_id UUID REFERENCES auth.users(id) ON DELETE SET NULL,
    tipo_documento VARCHAR(20) NOT NULL,
    documento VARCHAR(20) NOT NULL UNIQUE,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    fecha_nacimiento DATE NOT NULL,
    genero VARCHAR(10) NOT NULL,
    id_acudiente INT NOT NULL REFERENCES colegio.acudientes(id_acudiente),
    fecha_registro TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    estado_estudiante VARCHAR(20) DEFAULT 'Activo'
);

-- TABLA: colegio.matriculas
CREATE TABLE colegio.matriculas (
    id_matricula SERIAL PRIMARY KEY,
    codigo_matricula VARCHAR(50) NOT NULL UNIQUE,
    id_estudiante INT NOT NULL REFERENCES colegio.estudiantes(id_estudiante) ON DELETE CASCADE,
    id_periodo INT NOT NULL REFERENCES colegio.periodos_academicos(id_periodo),
    id_grupo INT NOT NULL REFERENCES colegio.grupos(id_grupo),
    fecha_matricula DATE NOT NULL DEFAULT CURRENT_DATE,
    estado_matricula VARCHAR(20) DEFAULT 'Matriculado',
    observaciones TEXT
);

-- TABLA: colegio.perfiles
CREATE TABLE colegio.perfiles (
    id SERIAL PRIMARY KEY,
    rol VARCHAR(50) NOT NULL,
    nombre_completo VARCHAR(200) NOT NULL,
    id_profesor INT REFERENCES colegio.profesores(id_profesor),
    id_acudiente INT REFERENCES colegio.acudientes(id_acudiente),
    id_estudiante INT REFERENCES colegio.estudiantes(id_estudiante),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
